زامل | منجزين المهمات

كلمات| #بسام_شانع
أداء| عيسى الليث
‏ •••••••••••••••
اليوم لا عاصم مـــن أهـــل العصيمات
ما مـــن عــــــذر قـــد با تقوم القيامة

عـــــاد الدواهي با تجـــــي والمُطِمات
عـــــاد النوازل في صواعـــــق غمامة

بني صريم تصـــــــــدرون الخضمّات
واتبنـــوا المــــــــوقف شديد الصرامة

ماعادها يالحلـــف الأرعـــن سلامات
تحــــــرم عليكم فـــــوق بقعا الإقامة

خارف صروف الدهـــــــر والمدلهمات
ظهرت ليـــــوم الفصـــــل آخر علامة

وانعـــم بحاشـــــد منجــزين المهمات
من فــوق سطـــح الأرض لاعند كامة

لو قد يشنوا خمـــس أو ســت هجمات
ما يمسوا إلَّا جـــوف قصــــر اليمامة

يتصــــدر الجبهــة شغاميـــــم وكمات
مشحـــون كمّـــــن راس يابس عدامة

تاريخهم معــروف في خمــس كلمات
تحــــرم على مـــــن حاربوه السلامة

يا اصحابنا لا يــرحم الله مــــــن مات
ومـــا سلك في العيـــش درب الكرامة

كان الصموت أوصافه أوصاف وسمات
واحــــــد رفـــــع عشرين مليون هامة

عودوا لسيرة عين رعـــــــــد المسمّات
في غرة التأريخ والدهــــــــــــــر شامة

مافي الأيادي يـــد مـــن غيــر بصمات
والراس ماهـــو راس مـــــن دون قامة

شمــــــــل القُبل يلتـــم عنـــــد الملمات
والكــــل وقـــــت الحــرب للكــــل لآمة

خــوض الوطيـــس مع آل طه وحومات
الحــــــــــــــق باتبقى صفــــة مستدامة

أسماء ترفعــــــــها عوامــــــــل وضمّات
وأسماء ميــــدان الشــــــــرف والكرامة

تسليــــم للبـــــــــــاري وغضبة لحرمات
الرب الأعلى فجـــــــــــــــــــروها قيامة

هبــــة مـــــــــع الله في مغــازي وكمّات
تشفي لظى كبــــــــــد النفوس المضامة

دمع البنيّات الصغــــــــــــــــار اليتيمات
داعي وجيـــــــه القبيـــــــــلة والشهامة

صريم ياخـــــارف عــــذر يالعصيمات
لا ترحموا حــــــد لـــــــــو تقطّع ندامة

ـــــــــــــــــــــــــــ
الإعلام الحربي 1442هـ
