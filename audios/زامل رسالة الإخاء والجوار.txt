زامل | رسالة الإخاء والجوار 

أداء / عيسى الليث
كلمات / أمين الجوفي
‏ •••••••••••••••
يــــا قـبـايـلنا بـشـبـه الـجـزيـرة
ويــن هـو حـق الاخـاء والـجوار

جــدنـا واحـــد وحــنـا عـشـيرة
والـشـواهد نـقـشها فــ الـحـجار

والـهـقـاوي لا جـهـتـكـم كـثـيـرة
والـكــرامــه فـــوق كــل اعـتـبار

عـهـدنـا فـيـكـم حـمـيه وغـيـرة
والخـوي لا اردوا خـويـه يــغـار

من رضي في صاحبه أو جويرة
بـالـمـجـازر والــعـبـث والــدمـار

بـايـجي لــه بـالـصغيرة صـغيرة
والــكـبـيـرة بـاتـجـي بـالــكـبـار

لــو يـصحي كـل واحـد ضـميرة
كـان مـاشعب الـيمن فـي حصار

كــان مــا سـلـمان نـافـخ بـكيرة
ذي يـــدق الـمـدرسـه والـمـطـار

والـمنازل والـعـجـوز الـكـبـيـرة
مسبحتهـا شـاهـده بـخــتـصـار

كــل مـا قـلنا عـليـه الـف خـيرة
انـتـفش ريــش الـزهـيمر وطـار

الـقـضـيـه لا سكـتوا خــطـيـرة
عـاقـبـتـهـا بــاتــصـل كـــــل دار

انـظـمـتكم بـيـد الاعــداء آجـيـرة
مـاتـريـد الـشـعـب يـمـلك قــرار

طـالـمـا امـريـكـا تـقـرر مـصـيرة
والله انـــه عـــاش ذلـــه وعـــار

بـالرُموت الـبيت الابـيض يديرة
مــــن يـمـيـنـه يـقـلـبـه لـلـيـسار

بـرمجة فـوق الـشعوب الـفقيرة
لـجـل مـن حـربة يـقش الـضمار

ذي يـبـيعة الاسـلحة والـذخيرة
يـحـرق أجـسـاد الـعيال الـصغار

من دمار اعمى? البصر والبصيرة
قــررت الأحــرار رمــي الــجمار

مـا نــسـيـنـا الاولــة والأخـيـرة
والـعـرب بــالـثـار تــأخــذ بـثـار

لا سـمعت الـشعب مـعلـن نفيرة
سـل ابـو جـبـريل سيف الـفـقار

تـاخـذك مـن كلـمـتـه قـشعريـرة
كـــن فــ اعـيـانة مـشـابيب نــار

مـاحـكمنا غـيـر مــن نـسـتخيرة
مـن كــتــاب الله خـط الـمـسـار

عـــز الأمــة فــ اتـبـاع الـمـسيرة
والــمــلازم والــعـلـم والــشـعـار
