رجال التصنيع

كلمات/ الحمزة المغربي
أداء/ عيسى الليث & محمد المحفدي
ـ〰️〰️〰️〰️〰️
جاهـِد ، صـنـّع ، ذخـّر ، طــوّر
جـهـدك لـلـه أكـــبـــر طـــاعــة

فاجئ وأدهِــش وأذهـل وأبهِر
أفــدي تصــنـــيـعـك وإبداعــه.

الـســاعـد جــنـب الـساعــد
في مــوقـفـنـا الـمـتـصـاعـد
يــا بــــارقـــنـــا والـــراعــد
ضــد إجــرام الصـهـيـونـيـّة

يا أشرف تصنيع ويا أعظم
قـــوّة عــُظمـــى تـتـــقــدّم
تــسـلــم أيــديـكـم تســلــم
يا صفــوة يمن الحــريــّة.

أطـلــق واعلق واحرِق واكسر
كبـــر الـمـسـتـكبـــر وأطمـاعَه

واضرب وأرعـب ، دمــّر ، فجّـر
لا يـهدأ صهـيونــي ســاعـــة.

أسوتـنــا الـمـخـتار مــحـمــّد
أشـــجـع قـائـد لـــلـعـُدة عـد
صَـنَـع السيـف ورمـح وشـرّد
بـ ســــلاحه جــمــع الـكـفــار

والسيف أصبح صــاروخــيـة
بــاليســتــي فـرط الصـوتـيــّة
تـضــرب والضـربـة صــدريـة
تـــتــحـــــدّى رصـــد و رادار

حـلــّق يـا مـســيـّرنــا حــلــّق
والـــغــازي دكـــّــه فــي داره‍

أشعـل نـارك واضرب من شق
وأرهـِـق صــفــّـارات إنـــذاره‍

بـالــزورق تـــلـــو الـزورق
نـبــحـِر والمـندب مـفـــرق
من يــتــحــدّانـا (يـــغـــرق )
بـيـد الـقـوات الـــبـحـريـة.

الفـتح الـمـوعــود أشـــرق
فـجـره‍ وأحـرزنـا الـبــيـرق
تــحـرق أمــريـكــا (تـحـرق)
وإســرائـيـل المـاسونــيــّة

جـــوّيـّــتـنــا بــأجــوانـــا
تبطش بـأسـراب أعــدانـا
والـكــافــر ما قـد عــانــىٰ
غيـر أوّل بأس الجــويــّـة.

يا فــرط الصــوت الـمــتــعـدد
رؤوس (المتشظـّي) في الهـيجـاء

(صعـّد) فرض حصارك صــــعــّد
واخـنــق بالـقــطـعـان المـلجــأ

ســبـــحـان الــبـاري مــن أيــّد.
ورمــى قــلـب الـكـفـر وســدد.
و هَدَى العـقل اليمني وأرشــد
لـــه تـصــنــيـع آخـــر إصــدار

اصنـع واحــنــا لـك بــا نـدعـم
كل الشعــب الـسند الأعــظـم
(أبـشـر) بــجـمــاجــمـنـا والـــدم
لـــو تــطــلـبـهــا مـــا نـحــتــار.

هيئة و رجــال التصـنـيـع
تنسـف أوهـام الـتـطبــيـع
في رأس الحربة ، وأقوى
صف جماعي مثل البنيان.

حمـداً لك يا مـولــى الجود
يا رب الــــفــتـح المـوعـود
أنت العون ومنـْك النصر
وفيك الــقـوة يـا رحمـٰن.
