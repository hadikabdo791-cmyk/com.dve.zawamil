زامل | طائر البشرى

عيسى الليث
كلمات | باسم مساوى
ـ•••••••••••••••
شدَى طــاير البـشرى وجــرّ أعـذب الألحان
بصـــــوتٍ تردد بـــين الأســماع وأطــــرَبها

وعمّ الفــرح فــي كــامل أرجــاء يمن الإيمان
وفـــزّت لهــــا الأرواح واعــــلت مـــراحبها

تراحــيب مـــادار الفَــلَك فـي مدى الأزمان
وتــــعداد نجـــــمات المــــجَرّ وكــــواكــبها

بذكرى لهــا قيـــمة عـــظيمه فــيّ الوجدان
ومحبّه جــرت في الدم من حُبّ صـــاحبها

بميــلاد خاتم الأنبيا المصـــطفى الــعدنان
وخــــيّر آل آدم واطــــهر الخــــلق وأنجَبها

محـــمّد رســـول الـلْه محمّد عظــيم الشّأن
مزلزل صـــروح أهــــل الضلال ومــــخربها

أتى ينــــقذ الإنســـان مـــن بطشة الإنسان
ويحيي القـــيم فــــي شـرق الأمّة ومغربها

ورحـــمة مـــن البـــاري لكـل إنسها والجان
ينـــــوّر بــــــــصايرها ويعـــلي مــــراتبـــها

من عــــبادة الشـــيطان لعبـــادة الرحــمن
الـــى كلـــــــمة الــــتوحيد لله ســـــار بها

رسول الــهدى والـــنور والعــدل والإحسان
بمــــــــيلاده الأنصـــــار تتـــــفاخر وتتــباها

تصــلّي علــــيك أرواح ياهــــاديّ الأكـــوان
عــلى مـــنهجك ســـارت وقـــامت بواجبها

بحــــبّك نغـــيض الكـــفر ونقــهقر العدوان
ونكــــسر خشــــوم أهــــل الضـلال ونأدّبها

ولو أمّة الإســــلام فـــي كامـــل الأوطـــان
يــــخافوا من أمــــريكا فحــن بانــحاربها

فلا خير فـــي أمّه غـــدت مـــا لها عـــنوان
وصـــــارت ضـــــحيّه لليـــهود ومـــــآربها

ولا خــير في أُمــــه تفرّط فـــــيّ الــــقرآن
وباعـــت قـــــــضيتها ونســــيَّت مــــناقبها

يمــــنّا بفــــضل الله والــــبدر يــــا مــــرّان
على منـــــهج المـــــختار ســــارت كـتايبها

وتشــــــهد مــــواقفنا من الــــماضيّ وللآن
صنــــعنا مــــلاحم عـزّت بعــــين طــــالبها
