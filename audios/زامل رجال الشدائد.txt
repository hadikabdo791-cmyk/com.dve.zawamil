زامل “رجال الشدائد”

أداء المنشد/ عيسى الليث

كلمات / أبوهادي الوايلي

ياســــلامي عليكــم يارجــال الشــدايــد
يـدنا الضــــــاربه بالضــــربه الحــــيدريه
يد طـــالوت تجــــازي كل عربــيد مــارد
تاخــــذ الثأر مــــن سافــك دمـنا البــريه
قل لقـــرقوش مخـزون الصواريخ وايــد
المــجـنـــح وبركـــان اليـــــماني هـــويه
طــوروهــا رجـال الله وأقرا الشـــواهــد
تعــــرف أن اليمن يملك مــواهـب قـويه
عاشـــت القوة العظمى وعـاش المجاهد
من صـــنع وأبتـكر صـاروخ من مدفعـيه
تضـــــربوا ثأركــم والفضــــل لله عــايـد
جل شـأنه صـــدق وعــــده ومكـن ولـيه
ياهـــل الكــبرياء والغطـــرسه والمــكايد
والله أن أنتصــار الـحـــق ســــنـه جلـيه
يالكـــروز المجـنح أضـــرب أبـــلاد زايـد
ترتــــدع من فعول البطش والعـنـجـهـيه
وأنـــت باالله يابـركـان شــــب الوقــايـد
في الرياض أشعل النيران الشمس حيه
أخــــذ بالـثأر للشــــعب اليـــماني عوايد
ثــأرنــا ناخــذه لو مـــــا تبـــــقى بقــــيه
حـــربـنا مامـعـكــم مــن وراها خــوايــد
بـالـوكـــاله تــقاتــــــلنا ولا لــك قـضـــيه
من تســــايد والامريكـي عليكـم تســايد
صمـــت لتــرامب وأسرائيل مثل المطيه
ســيدي ســـيدي يا قايدي ياخيــر قايـد
رؤســــنا دون راسك يا أبن طـــه فــديه
‏
