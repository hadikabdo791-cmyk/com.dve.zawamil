يـــاسـيد الكـونين جــــــات المــــلايين
تـــعلـن ولاهـــــا لك على روس الأشهاد

لك تخـــضع الدنيا مـــلوك وســــلاطين
ولك فـــي ضـمير الــكون ايـات واعـياد

خـــتم النــبوة والســـــنن والقــــــوانين
صـــلـــى عليك الله مـــــاحن رعـــــــــاد

بك نحتفل في سـاعة العســـــرواللـــين
بك ننتــصر فـي يـــــوم حــامي ووقــاد

اخـرجتـــنا مـــن ظلمة الجــهل للــــدين
أمــة عظيمة مـجدها فـــوق الامـــــجاد

حنـا اليمن اهــــــل القــــلوب المــيامين
حــــــــــنا لـــدين الله قـــادات واوتـــاد

هـاتوا الوجود وهـــاتوا اشقى المعادين
نـبري غــليل الــــمـوت عـادات الاجـواد

خــــــضنا لضــاها والمعـــارك بـــراكين
يـوم الندى جينا جمــــاعــات وافــــــراد

رايــــاتنا فـــي بـدر وســــهول صــــفين
هـــي هـــذي الــرايات ياحـــلف الاحـقاد

فيها عـلـي والمــصطفــى وال يـاسيــن
فيــــها الــرجال اللـي مــن العـزم تزداد
