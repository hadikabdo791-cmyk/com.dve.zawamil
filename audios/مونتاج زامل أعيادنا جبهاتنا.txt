#أعيادنا_جبهاتنا
ـ•••••••••••••••

يـاقـبـيلة داعـــي نـكـف زمــجـر وكـلـن يـسمعه
الــعـيـد فـالـجـبـهات من حيث النمارة والسباع

حــيـث الـكـرامـة والـشـهامة والـقـيم مـتـربعة
عـلـى شـمـوخ الـعـز يـحـميها الـمظلع والـكراع

أعــيـادنـا جـبـهـاتـنا كــلــن وعــيــده مـوقـعـة
قــوى الـعـدو مـتأمرة نـفشل حـيَلهم والخداع

قـال الـمثل ذي مايقايس قبل يقطع ما أخضعه
مــعـاد بـعـد الـقـطع شـبـره بـايـفيده والــذراع

والـشط لا اتـواسع من الصعب ان الأبره ترقعه
والـقـافـلة لا فـكَّـهـا الـجـمـَّال تـاكـلـها الـضـباع

الــبــريـه مــــن جـيـشـنـا ولـجـانـنـا مــتـوزعـة
فــي كــل جـبـهة مـستعدة بـالمهاجم والـدفاع

والـمـدفـعـيه جــاهــزه كــلــن مَـعَـمِّـر مـدفـعـه
والـقـوة الـصـاروخية تـنـصب مـنصات الـوداع

والـبـحرية حول الـشـريط الـساحلى مـتجمعة
مـن سـاقتة نـفسة سـوا فـالبحر والا الـبر ضاع

وابـلـيس مـايـقدر يـوطـى مـن قـد الله ارفـعه
مـهما يـحاول يـخدعه يـفشل ولـو يلبس قناع

مهما طغى الظالم يجى له يوم يلقى مصرعه
بالصاع يلقى من رجاجيل اليمن خمسين صاع

قـولـوا لـحـلف الـشـر يـخـسى مـاحـدا بـايـنفعه
لـوكان فـيهم خـير فكوا أسر الآقصى والقطاع

أحـنا العرب الاقحاح واحنا الدين واحنا منبعه
صم الجبل يبقى جبل شامخ وتبقى القاع قاع
