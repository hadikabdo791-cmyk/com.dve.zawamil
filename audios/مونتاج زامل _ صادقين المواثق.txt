زامل | صادقين المواثق

عيسى الليث
كلمات / باسم مساوى
ـ•••••••••••••••
باذلـين النــفوس وصـــــانعين الــــفَــوارِق
من عطاهم على درب العطاء في خِضَمه

لاح بــــــرّاق ذكـرى صـــادقين الـــــمَواثِق
مَن عطاهم علـــى درب العطا في خــضمّه

كلّ مــــــؤمن مع الرحمـــن كانه مســــابِق
في سبــــيله بذل روحـــه وله ســـاق دمّه

باذلـين النــفوس وصـــــانعين الــــفَــوارِق
مخلــــصين العــــهود ومنـجزين المــــهمّة

هـــكذا طـــبع من هو كـــان فالله واثـــــق
عــزمهم عـــــزم مـــــا ينهـــدّ يوم الملـــمّة

وبـــدماهم بنوا لســــلامهم صـــرح شاهِق
واركَـــزوا بــيرق الأنصـــار فــي كلّ قـــمّة

عــهد منا مـــا نساوم فـي دمــــا كـلّ صادق
وثارهـــــم والــوفا مـــــلزوم فــي كلّ ذِمّة

لا ورب المـغارب لا ورب المـشارق
عـــاد فالجـــسم روح وعـــاد فـالروح هِمّة

ما لذا الحلف الأرعن غـــير حــارق وخارق
فـــي رجا ليـــــلةٍ لا جـــوف داره نـــــكمّه

كم مــــع الله ودّعـــــــــنا وداع المــــفارق
هــــذا اخــــوه وهـذاك ابنه وذاك ابن عمّه

عـــاد للتضــــحية مليـــون فـــارق وفـارق
ودمّنــــا سيــل بايجــــرف ســعود ويطمّه

دام والكـــون واللـــي فــــيه لاشـكّ نــافِق
حــي نفـــح الشــــهادة والخلد ياحي شمّه

ســـيّدي كــــلّنا بيـــدك سيـــوف وبنـــادق
اللــــــقا يــــومه ان ســــمّوه والا فــــسمّه

ابتـــــدينا مـــراسم نصـــــرنا بالمـــــحارق
وبالشـــــهادة لنـــيل الــفوز الاكـــبر نـتِـمّه
