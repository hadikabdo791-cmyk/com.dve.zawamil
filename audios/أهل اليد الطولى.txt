أهل اليد الطولى

عيسى الليث
كلمات| بسام شانع

اذا بَذَلْنَـــــا بَذْلَنَــــــا البــــــذل الأجل
إذا اخَـــذْنَا أخْـــذَنَا الأخــــــذ الوبيل

يـا رزح ظهـــــري يـا جبل يرزح جبل
وجه الرضى ثلثين مـن راس الصميل

مــن له ثِقَل غيــر الـــذي مـــا له ثِقَل
واهل اليد الطولى لها البـــاع الطويل

اذا بَذَلْنَـــــا بَذْلَنَــــــا البــــــذل الأجل
إذا اخَـــذْنَا أخْـــذَنَا الأخــــــذ الوبيل

اذا بَذَلْنَـــــا بَذْلَنَــــــا البــــــذل الأجل
إذا اخَـــذْنَا أخْـــذَنَا الأخــــــذ الوبيل

إذا دفعنـــا نـــــدفع الخطـــب الجلل
واحنا خفـاف الظـل فـ اليــوم الثقيل

واكثــــر معونتــنـا مؤونتهــــــا الأقل
ومــع بساطتنــا نجيــــب المستحيل

إذا دفعنـــا نـــــدفع الخطـــب الجلل
واحنا خفـاف الظـل فـ اليــوم الثقيل

يلجـــا لنـــــا عالـــــم وتخشانــا دول
ومــرادنـــا سامــي ومــسعانــــا نبيل

إذا رددنــــــا ردَّنــــــا مثنــــــــي دبل
وذا شَكَــرْنا شُكْـــرَنا الشكــــر الجزيل

إذا رددنــــــا ردَّنــــــا مثنــــــــي دبل
وذا شَكَــرْنا شُكْـــرَنا الشكــــر الجزيل

كــــف الـــــذي لا قـــال با يفعل فَعَلْ
تسبـــــق لسانـــه يختصـر والا يطيل

مهمـا حـدث مهمـا جـرى مهمـا حصل
ما يختلـــــف او ينحـــرف والا يميل

والغـــرب هو والشـرق يدخل لا دخل
في حــرب يخـــرج منها السالم قتيل

نكال مــــوت ازول قضــا حتمي اجل
لا جـــا  فلا منجـــى ولا ملجــى يقيل

عـــز الــذي ذل القــــوى العظمى وجل
حسب الــــذي اتوكل على نعـم الوكيل

للحـــــارق الخـــــارق سنـــــا بارق امل
والصاعــق الساحــق فنا ماحـــق مزيل

وابــل كســـف وابــل ظلل تـردف ظلل
يا الــداهم الهـــادم ويا الهـــول الهويل

نزل عــذاب الحاطـــــم الناقـــــــم نزل
كــان ابرهة بـ الفيل أو مــــن دون فيل
