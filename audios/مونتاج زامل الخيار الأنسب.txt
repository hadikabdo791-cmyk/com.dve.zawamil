الخيار الأنسب

عيسى الليث – رشاد الخزان
كلمات | بسام شانع
ـ•••••••••••••••
العالم دخـــــل جحـــر الضب
والحنبه دخـــــــــل بالمقلوب

خلــــف ام الكبايـــــــر والأب
يتقاطـــــــر عيـــــال الانبوب

قــــــد نغبش بيــــوت الحُرَّب
مـن يبحث على اجباح النوب

والحنبه بحجـــــــــم المحنب
والنكبة بحجـــــــــم المنكوب

واضـيـَــق يا مضيــق المندب
مـــن ســــم الخياط المثقوب

واصعب مــن جبال الجرشب
واصلب مــن جبال المصلوب

واشمخ مــن شوامخ شرعب
واشمخ مــن شوامخ كهبوب

والمعجب بضيقــــك معجب
والمرعوب منـــــــك مرعوب

يـــــا يورانيـــــــوم اتخصَّب
يـــــا مفعول مطلق منصوب

يا الجيـــش الــــذي لا يغلب
ياابن اضرى معارك وحـروب

سطح البحر الاحمــر قشبب
من سـوط العذاب المصبوب

والهنـــــدي والابيــــض لبّب
مـــن نــــص البيان المكتوب

لا يمكــــــن يمــــــــرّر مركب
حجمه حجم ياجورة طوب

ســــره ســـــر مــــــا يتسرّب
يتفاعل مــــــع المـــا ويذوب

برج البـــــــــــث برج العقرب
قطب الملتقــــط والمجذوب

والراصــد مقعّــــــــر واحدب
ثاقـــــب يرقــــب اللامرقوب

يقتــــــص الأثــــــــر يتعقَّب
يفحص فحص بالمكروسكوب

يحسب للقيــــــــادة محسب
كـــم في كـــل وطلة مكروب

كن البحــــر داخـــــل مضرب
عمقــه مثلمـــا عمـــق الكوب

سطحه مثـــل سطح المكتب
في شاشة جهـــــاز الحاسوب

يختــــــار الخيــــــار الأنسب
حسب افضل طريقة وأسلوب

ويضـــع كـــــــل مــــا يتطلب
وضعه في المكــــان المطلوب

والطـــــوفان طـــــوفان الرب
لا ازداد ارتفــــــــاع المنسوب

لا اتضعضــع ورا مـــا اتذبذب
واختـــض الجبــــان الرعبوب

موت ازوب علــى موت ازوب
مـــن راس الرجــــــا للعركوب

لو عالم جنــــــــــوب الكوكب
مثـــل الجــــن عالم محجوب

والمعبــــــر ظــــــلام الغيهب
والعــــابر خفــــــايــا وغيوب

موت ازوب علــى موت ازوب
مـــن راس الرجــــــا للعركوب
